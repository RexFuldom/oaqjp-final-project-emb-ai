from flask import Flask, render_template, request
from EmotionDetection.emotion_detection import emotion_detector

# Iniciar la aplicación Flask
app = Flask("Emotion Detector")

@app.route("/emotionDetector")
def sent_analyzer():
    # Obtener el texto proporcionado a través de los argumentos de la URL (parámetro 'textToAnalyze')
    text_to_analyze = request.args.get('textToAnalyze')
    
    # Llamar a la función emotion_detector del paquete creado
    response = emotion_detector(text_to_analyze)
    
    # Comprobar si la emoción dominante es None (manejo de errores/texto vacío)
    if response['dominant_emotion'] is None:
        return "Invalid text! Please try again!"
    
    # Extraer los valores individuales
    anger = response['anger']
    disgust = response['disgust']
    fear = response['fear']
    joy = response['joy']
    sadness = response['sadness']
    dominant_emotion = response['dominant_emotion']
    
    # Formatear la salida exactamente como lo requiere el cliente
    return (
        f"For the given statement, the system response is 'anger': {anger}, "
        f"'disgust': {disgust}, 'fear': {fear}, 'joy': {joy} and "
        f"'sadness': {sadness}. The dominant emotion is <b>{dominant_emotion}</b>."
    )

@app.route("/")
def render_index_page():
    return render_template('index.html')

if __name__ == "__main__":
    # Desplegar la aplicación en localhost:5000
    app.run(host="0.0.0.0", port=5000)
